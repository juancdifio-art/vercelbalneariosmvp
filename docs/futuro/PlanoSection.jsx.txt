import React, { useMemo, useState } from 'react';
import { addDays } from 'date-fns';
import { format } from '../lib/dates';
import { ServiceIcon } from './icons';

/**
 * Plano del balneario: la planta entera de un día, a pantalla completa.
 *
 * A diferencia de la Vista rápida —que muestra un servicio por vez dentro del
 * panel— acá se ve todo junto y en grande, porque la pregunta que contesta es
 * "cómo está la playa hoy". Por eso rompe el recuadro con márgenes negativos
 * y ocupa el alto de la ventana.
 *
 * La disposición todavía es una grilla regular. Cuando esté cerrada la planta
 * relevada de Zeus, cada unidad pasa a dibujarse en su posición real y el
 * resto de la pantalla queda igual.
 */
function PlanoSection({
  establishment,
  carpasReservations,
  sombrillasReservations,
  parkingReservations,
  reservationGroups,
  onViewReservationDetails,
  onOpenNewCarpaReservation,
  onOpenNewSombrillaReservation,
  onOpenNewParkingReservation
}) {
  const [dayOffset, setDayOffset] = useState(0);

  const today = useMemo(() => new Date(), []);
  const currentDay = useMemo(() => addDays(today, dayOffset), [today, dayOffset]);
  const currentDateStr = format(currentDay, 'yyyy-MM-dd');

  const services = useMemo(
    () =>
      [
        establishment?.hasCarpas && {
          id: 'carpas',
          serviceType: 'carpa',
          label: 'Carpas',
          prefijo: 'Carpa',
          capacity: Number.parseInt(establishment.carpasCapacity ?? '0', 10) || 0,
          reservations: carpasReservations,
          onNueva: onOpenNewCarpaReservation
        },
        establishment?.hasSombrillas && {
          id: 'sombrillas',
          serviceType: 'sombrilla',
          label: 'Sombrillas',
          prefijo: 'Sombrilla',
          capacity: Number.parseInt(establishment.sombrillasCapacity ?? '0', 10) || 0,
          reservations: sombrillasReservations,
          onNueva: onOpenNewSombrillaReservation
        },
        establishment?.hasParking && {
          id: 'parking',
          serviceType: 'parking',
          label: 'Estacionamiento',
          prefijo: 'Plaza',
          capacity: Number.parseInt(establishment.parkingCapacity ?? '0', 10) || 0,
          reservations: parkingReservations,
          onNueva: onOpenNewParkingReservation
        }
      ].filter(Boolean),
    [
      establishment,
      carpasReservations,
      sombrillasReservations,
      parkingReservations,
      onOpenNewCarpaReservation,
      onOpenNewSombrillaReservation,
      onOpenNewParkingReservation
    ]
  );

  const ocupadas = (service) => {
    let n = 0;
    for (let i = 1; i <= service.capacity; i += 1) {
      if (service.reservations?.[`${currentDateStr}-${i}`]) n += 1;
    }
    return n;
  };

  const abrirUnidad = (service, numero, estaOcupada) => {
    if (estaOcupada) {
      if (!onViewReservationDetails || !Array.isArray(reservationGroups)) return;
      const grupo = reservationGroups.find(
        (g) =>
          g.serviceType === service.serviceType &&
          g.resourceNumber === numero &&
          g.status === 'active' &&
          g.startDate <= currentDateStr &&
          g.endDate >= currentDateStr
      );
      if (grupo) onViewReservationDetails(grupo);
      return;
    }
    if (service.onNueva) service.onNueva(numero, currentDay);
  };

  const esHoy = dayOffset === 0;

  return (
    <div className="-m-5 flex min-h-[calc(100vh-5rem)] flex-col overflow-hidden rounded-2xl bg-gradient-to-b from-sky-50 to-amber-50/60 sm:-m-7">
      {/* Barra del día */}
      <div className="flex flex-wrap items-center gap-3 border-b border-cyan-100 bg-white/80 px-4 py-3 backdrop-blur sm:px-6">
        <div className="flex items-center gap-2">
          <button
            type="button"
            onClick={() => setDayOffset((p) => p - 1)}
            className="inline-flex h-8 w-8 items-center justify-center rounded-lg border border-cyan-300 bg-white text-cyan-700 transition hover:bg-cyan-50 focus:outline-none focus-visible:ring-2 focus-visible:ring-cyan-500"
            aria-label="Día anterior"
          >
            ◀
          </button>
          <button
            type="button"
            onClick={() => setDayOffset(0)}
            disabled={esHoy}
            className={
              'rounded-lg border px-3 py-1.5 text-[13px] font-medium transition focus:outline-none focus-visible:ring-2 focus-visible:ring-cyan-500 ' +
              (esHoy
                ? 'border-slate-200 bg-slate-100 text-slate-400'
                : 'border-cyan-300 bg-white text-cyan-700 hover:bg-cyan-50')
            }
          >
            Hoy
          </button>
          <button
            type="button"
            onClick={() => setDayOffset((p) => p + 1)}
            className="inline-flex h-8 w-8 items-center justify-center rounded-lg border border-cyan-300 bg-white text-cyan-700 transition hover:bg-cyan-50 focus:outline-none focus-visible:ring-2 focus-visible:ring-cyan-500"
            aria-label="Día siguiente"
          >
            ▶
          </button>
        </div>

        <p className="text-[15px] font-semibold tabular-nums text-slate-900">
          {format(currentDay, "EEEE d 'de' MMMM")}
        </p>

        <div className="ml-auto flex flex-wrap items-center gap-2">
          {services.map((service) => {
            const usadas = ocupadas(service);
            const pct = service.capacity ? Math.round((usadas / service.capacity) * 100) : 0;
            return (
              <span
                key={service.id}
                className="inline-flex items-center gap-2 rounded-full border border-slate-200 bg-white px-3 py-1.5 text-[12px] text-slate-700"
              >
                <ServiceIcon serviceId={service.id} className="h-4 w-4 shrink-0 text-slate-500" />
                <span className="font-medium">{service.label}</span>
                <span className="font-mono tabular-nums text-slate-900">
                  {usadas}/{service.capacity}
                </span>
                <span className="tabular-nums text-slate-500">{pct}%</span>
              </span>
            );
          })}
        </div>
      </div>

      {/* Referencias */}
      <div className="flex flex-wrap items-center gap-x-5 gap-y-2 px-4 py-2 text-[11px] text-slate-600 sm:px-6">
        <span className="inline-flex items-center gap-1.5">
          <span className="h-3 w-3 rounded-sm border border-emerald-700 bg-emerald-600" />
          Ocupada
        </span>
        <span className="inline-flex items-center gap-1.5">
          <span className="h-3 w-3 rounded-sm border border-slate-300 bg-white" />
          Libre
        </span>
        <span className="text-slate-400">Clic en una unidad para reservarla o ver su reserva.</span>
      </div>

      {/* Planta */}
      <div className="flex-1 space-y-4 overflow-auto px-4 pb-6 sm:px-6">
        {services.map((service) => {
          if (!service.capacity) return null;
          const unidades = Array.from({ length: service.capacity }, (_, i) => i + 1);

          return (
            <section key={service.id} className="rounded-xl border border-slate-200 bg-white/90 p-3 sm:p-4">
              <h2 className="mb-2 flex items-center gap-2 text-sm font-semibold text-slate-900">
                <ServiceIcon serviceId={service.id} className="h-4 w-4 shrink-0 text-slate-500" />
                {service.label}
                <span className="font-mono text-[11px] font-normal tabular-nums text-slate-500">
                  {service.capacity} unidades
                </span>
              </h2>

              <div className="grid grid-cols-[repeat(auto-fill,minmax(46px,1fr))] gap-1.5">
                {unidades.map((numero) => {
                  const estaOcupada = Boolean(service.reservations?.[`${currentDateStr}-${numero}`]);
                  return (
                    <button
                      key={numero}
                      type="button"
                      onClick={() => abrirUnidad(service, numero, estaOcupada)}
                      title={`${service.prefijo} ${numero} — ${estaOcupada ? 'ocupada' : 'libre'}`}
                      className={
                        'flex h-9 items-center justify-center rounded-md border text-[12px] font-medium tabular-nums transition focus:outline-none focus-visible:ring-2 focus-visible:ring-cyan-500 ' +
                        (estaOcupada
                          ? 'border-emerald-700 bg-emerald-600 text-white hover:bg-emerald-700'
                          : 'border-slate-300 bg-white text-slate-700 hover:border-cyan-400 hover:bg-cyan-50')
                      }
                    >
                      {numero}
                    </button>
                  );
                })}
              </div>
            </section>
          );
        })}

        {services.length === 0 && (
          <p className="py-10 text-center text-sm text-slate-500">
            Este establecimiento todavía no tiene servicios configurados.
          </p>
        )}
      </div>
    </div>
  );
}

export default PlanoSection;
